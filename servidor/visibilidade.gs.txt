// todos os ligados por pai, mãe ou casamento (amigo/namorado aceito tem visão limitada: visaoAmigos_)
function visiveis_(all, startId) {
  const seen = new Set(); if (!startId) return seen;
  const byId = {}; all.forEach(p => byId[p.id] = p);
  const adj = {}; const add = (a, b) => { if (!a || !b || !byId[a] || !byId[b]) return; (adj[a] = adj[a] || []).push(b); (adj[b] = adj[b] || []).push(a); };
  all.forEach(p => { add(p.id, p.data.father); add(p.id, p.data.mother); add(p.id, p.data.spouse); });
  seen.add(startId); const q = [startId];
  while (q.length) { const x = q.pop(); (adj[x] || []).forEach(y => { if (!seen.has(y)) { seen.add(y); q.push(y); } }); }
  return seen;
}
function norm_(t) { return String(t || '').normalize('NFD').replace(/[̀-ͯ]/g, '').toLowerCase().replace(/[^a-z ]/g, ' ').replace(/\b(da|de|do|das|dos|e)\b/g, ' ').replace(/\s+/g, ' ').trim(); }
function salvarPessoa_(id, data) {
  const s = aba_('people'), ids = s.getRange(1, 1, Math.max(1, s.getLastRow()), 1).getValues().map(x => String(x[0]));
  const linha = [id, JSON.stringify(data), new Date()], i = ids.indexOf(String(id));
  if (i > 0) s.getRange(i + 1, 1, 1, 3).setValues([linha]); else s.appendRow(linha);
}
// procura pelo nome que a pessoa digitou (só devolve quem combina com o que ela escreveu)
function buscar_(all, nome) {
  const q = norm_(nome); if (q.length < 2) return [];
  const qt = q.split(' ');
  const yr = d => { const m = String(d || '').match(/(\d{4})/); return m ? m[1] : ''; };
  const ini = id => { const p = all.find(x => x.id === id); return p && !p.data.unnamed ? p.data.name.trim().charAt(0).toUpperCase() + '.' : ''; };
  return all.filter(p => !p.data.unnamed).map(p => {
    const nt = norm_(p.data.name).split(' '), ap = norm_(p.data.apelido);
    let sc = 0; if (nt[0] === qt[0] || ap === q || ap === qt[0]) sc += 2;
    qt.slice(1).forEach(w => { if (nt.includes(w)) sc += 1; });
    if (norm_(p.data.name) === q) sc += 3;
    return { p, sc };
  }).filter(x => x.sc >= 2).sort((a, b) => b.sc - a.sc).slice(0, 3).map(x => {
    const d = x.p.data, pais = [ini(d.father), ini(d.mother)].filter(Boolean).join(' e ');
    return { id: x.p.id, name: d.name, ano: yr(d.birth), dica: pais ? 'filho(a) de ' + pais : '' };
  });
}
function novoId_() { return 'p' + Date.now().toString(36) + Math.random().toString(36).slice(2, 7); }
// liga quem acabou de entrar à árvore, aqui no servidor (a pessoa ainda não enxerga a árvore)
function vincular_(r, a) {
  const c = cadLinha_(r.code); if (!c) throw new Error('Complete o cadastro antes.');
  const cad = c.data, all = pessoas_(), byId = {}; all.forEach(p => byId[p.id] = p.data);
  const rel = a.rel, to = a.to ? byId[a.to] : null;
  if (rel !== 'conhecer' && !to) throw new Error('Pessoa não encontrada.');
  if (cad.personId && byId[cad.personId]) throw new Error('Você já está ligado(a) à árvore.');
  if (rel === 'eu' && aba_('cadastro').getDataRange().getValues().slice(1).some(x => String(x[11]) === String(a.to))) throw new Error('Essa pessoa já tem cadastro próprio.');
  let meId;
  if (rel === 'eu') {
    meId = a.to; const me = byId[meId]; me.provisional = false; if (cad.nascimento && !me.birth) me.birth = cad.nascimento;
    me.updatedBy = r.uid; me.updatedByName = cad.nome.split(' ')[0]; me.updatedAt = Date.now(); salvarPessoa_(meId, me);
  } else {
    meId = novoId_();
    const me = { name: cad.nome, sex: a.sexo || '', father: null, mother: null, parents: [], spouse: null, photo: null, birth: cad.nascimento || '', deceased: false, death: '', apelido: '', provisional: true, createdAt: Date.now(), createdBy: r.uid, createdByName: cad.nome.split(' ')[0] };
    if (rel === 'filho') {
      if (to.sex === 'F') me.mother = a.to; else me.father = a.to;
      const sp = to.spouse && byId[to.spouse];
      if (sp) { if (sp.sex === 'F' && !me.mother) me.mother = to.spouse; else if (!me.father) me.father = to.spouse; }
    } else if (rel === 'irmao') { me.father = to.father || null; me.mother = to.mother || null; }
    else if (rel === 'conjuge') { if (to.spouse) throw new Error(to.name + ' já tem cônjuge cadastrado.'); me.spouse = a.to; to.spouse = meId; salvarPessoa_(a.to, to); }
    else if (rel === 'pai') {
      if (cad.sexo === 'F' || a.sexo === 'F') { if (to.mother) throw new Error(to.name + ' já tem mãe cadastrada.'); to.mother = meId; me.sex = 'F'; }
      else { if (to.father) throw new Error(to.name + ' já tem pai cadastrado.'); to.father = meId; me.sex = me.sex || 'M'; }
      to.parents = [to.father, to.mother].filter(Boolean); salvarPessoa_(a.to, to);
    }
    // quem veio por convite já está confirmado por quem convidou
    else if (rel === 'amigo' || rel === 'namorado') { me.links = [{ type: rel, to: a.to, confirmado: !!a.convite, convite: !!a.convite }]; }
    me.parents = [me.father, me.mother].filter(Boolean);
    salvarPessoa_(meId, me);
  }
  cad.personId = meId; const s = aba_('cadastro'); s.getRange(c.row, 12).setValue(meId); s.getRange(c.row, 14).setValue(JSON.stringify(cad));
  return { personId: meId };
}
// confirmação: antes de ligar, faz 2 perguntas de parentesco que quem é da família sabe responder
function previa_(a) {
  const all = pessoas_(), byId = {}; all.forEach(p => byId[p.id] = p.data);
  const X = byId[a.to]; if (!X) throw new Error('Pessoa não encontrada.');
  const kids = id => all.filter(p => (p.data.father === id || p.data.mother === id) && !p.data.unnamed).map(p => p.data.name);
  const sibs = d => all.filter(p => p.data !== d && !p.data.unnamed && ((d.father && p.data.father === d.father) || (d.mother && p.data.mother === d.mother))).map(p => p.data.name);
  const pais = d => [d.father, d.mother].map(id => byId[id]).filter(p => p && !p.unnamed).map(p => p.name);
  const Q = [], put = t => { if (t && Q.length < 2 && !Q.includes(t)) Q.push(t); };
  const sp = X.spouse && byId[X.spouse] && !byId[X.spouse].unnamed ? byId[X.spouse].name : '';
  if (a.rel === 'filho') { kids(a.to).slice(0, 1).forEach(k => put('Então você é irmão/irmã de ' + k + '?')); pais(X).slice(0, 1).forEach(n => put('E é neto(a) de ' + n + '?')); if (sp) put('Sua mãe/seu pai é ' + sp + '?'); sibs(X).slice(0, 1).forEach(n => put('E ' + n + ' é seu tio/sua tia?')); }
  else if (a.rel === 'irmao') { pais(X).forEach(n => put('Então você é filho(a) de ' + n + '?')); sibs(X).slice(0, 1).forEach(n => put('E também é irmão/irmã de ' + n + '?')); }
  else if (a.rel === 'conjuge') { kids(a.to).slice(0, 1).forEach(k => put('Então ' + k + ' é seu filho/sua filha?')); pais(X).slice(0, 1).forEach(n => put('E ' + n + ' é seu sogro/sua sogra?')); sibs(X).slice(0, 1).forEach(n => put('E ' + n + ' é seu cunhado/sua cunhada?')); }
  else if (a.rel === 'eu') { pais(X).slice(0, 1).forEach(n => put('Você é filho(a) de ' + n + '?')); sibs(X).slice(0, 1).forEach(n => put('E é irmão/irmã de ' + n + '?')); if (sp) put('É casado(a) com ' + sp + '?'); kids(a.to).slice(0, 1).forEach(k => put(k + ' é seu filho/sua filha?')); }
  return { perguntas: Q };
}

// pedidos de namorado(a)/amigo(a) esperando a resposta da pessoa (ou do dono)
function pedidos_(r) {
  const all = pessoas_(), c = cadLinha_(r.code), eu = c && c.data.personId;
  const out = [];
  all.forEach(p => (p.data.links || []).forEach(l => {
    if (!l.confirmado && !l.recusado && (r.role === 'owner' || l.to === eu)) {
      const alvo = all.find(x => x.id === l.to);
      out.push({ de: p.id, nome: p.data.name, tipo: l.type, para: l.to, paraNome: alvo ? alvo.data.name : '' });
    }
  }));
  // amigo/namorado já aceito pedindo para ver os filhos de quem o aceitou
  if (eu) { const byId = {}; all.forEach(p => byId[p.id] = p); let cads = null;
    all.forEach(p => (p.data.links || []).forEach(l => { if (!l.confirmado || !l.pedeFilhos || !byId[l.to]) return; const T = byId[l.to];
      all.forEach(C => { if (C.data.father !== T.id && C.data.mother !== T.id) return; cads = cads || comCadastro_();
        const q = quemAutoriza_(byId, T, C, cads) || [], ok = (l.ver || {})[C.id] || [], nao = (l.naoVer || {})[C.id] || [];
        if (q.includes(eu) && !ok.includes(eu) && !nao.includes(eu)) out.push({ de: p.id, nome: p.data.name, tipo: 'verFilho', para: T.id, paraNome: T.data.name, filho: C.id, filhoNome: C.data.name, eProprio: C.id === eu, menor: menor_(C.data) });
      }); })); }
  return out;
}
function responder_(r, a) {
  const c = cadLinha_(r.code), eu = c && c.data.personId, all = pessoas_();
  const p = all.find(x => x.id === a.de); if (!p) throw new Error('Pedido não encontrado.');
  if (a.filho) { // autorização para ver um filho
    const l = (p.data.links || []).find(x => x.to === a.para && x.confirmado); if (!l) throw new Error('Pedido não encontrado.');
    const byId = {}; all.forEach(x => byId[x.id] = x); const C = byId[a.filho], T = byId[a.para];
    const q = (C && T && quemAutoriza_(byId, T, C, comCadastro_())) || []; if (!q.includes(eu)) throw new Error('Só os pais (ou o próprio filho, se maior) podem autorizar.');
    const k = a.aceitar ? 'ver' : 'naoVer'; l[k] = l[k] || {}; l[k][a.filho] = (l[k][a.filho] || []).concat(eu);
    salvarPessoa_(p.id, p.data); return { ok: true };
  }
  if (a.pedeFilhos) { // o amigo pede para ver os filhos de quem o aceitou
    if (p.id !== eu) throw new Error('Só o próprio amigo pode pedir.');
    const l = (p.data.links || []).find(x => x.to === a.para && x.confirmado); if (!l) throw new Error('Ainda não aceito.');
    l.pedeFilhos = true; salvarPessoa_(p.id, p.data); return { ok: true };
  }
  const l = (p.data.links || []).find(x => x.to === a.para && !x.confirmado);
  if (!l) throw new Error('Pedido não encontrado.');
  if (r.role !== 'owner' && l.to !== eu) throw new Error('Só a própria pessoa pode responder.');
  if (a.aceitar) { l.confirmado = true; l.confirmadoEm = Date.now(); if (a.filhos) { l.pedeFilhos = true; const T = all.find(x => x.id === l.to); all.forEach(C => { if (C.data.father === l.to || C.data.mother === l.to) { l.ver = l.ver || {}; l.ver[C.id] = (l.ver[C.id] || []).concat(eu || l.to); } }); } } else { l.recusado = true; }
  salvarPessoa_(p.id, p.data);
  return { ok: true };
}

// quem mandou o convite: só o primeiro nome, para a tela de boas-vindas
function convidou_(a) { const p = pessoas_().find(x => x.id === String(a && a.id)); return p && !p.data.unnamed ? { nome: (p.data.apelido || p.data.name.split(' ')[0]) } : null; }
// amigo/namorado aceito: vê o globo de quem o aceitou, só com os nomes dela, do cônjuge e dos pais;
// os filhos só aparecem com autorização (menor: os pais; maior: o pai/mãe e o próprio filho). O resto vem vazio.
// A família de quem aceitou vê o amigo ao lado, só com o nome.
function menor_(d) { if (!d || d.deceased) return false; if (!d.birth) return !!d.minor; const b = new Date(d.birth + 'T12:00:00'), t = new Date(); let a = t.getFullYear() - b.getFullYear(); if (t.getMonth() < b.getMonth() || (t.getMonth() === b.getMonth() && t.getDate() < b.getDate())) a--; return a < 18; }
function comCadastro_() { return new Set(aba_('cadastro').getDataRange().getValues().slice(1).map(x => String(x[11])).filter(Boolean)); }
// quem precisa autorizar o amigo de T a ver o filho C
function quemAutoriza_(byId, T, C, cads) {
  const pais = [C.data.father, C.data.mother].filter(Boolean);
  if (menor_(C.data)) return pais.filter(id => id === T.id || cads.has(id));
  return cads.has(C.id) ? [T.id, C.id] : null; // maior sem cadastro ainda não pode autorizar
}
function filhoLiberado_(byId, T, C, l, cads) {
  const q = quemAutoriza_(byId, T, C, cads); if (!q) return false;
  const ok = ((l.ver || {})[C.id] || []); return q.every(id => ok.includes(id));
}
function visaoAmigos_(all, eu, vis) {
  const byId = {}; all.forEach(p => byId[p.id] = p);
  const out = {}, put = (p, nivel) => { if (vis.has(p.id) || (out[p.id] && out[p.id].nivel === 'nome')) return; out[p.id] = { p, nivel }; };
  const limpa = (p, nivel) => { const d = p.data, r = { name: '', sex: d.sex || '', father: d.father || null, mother: d.mother || null, parents: [d.father, d.mother].filter(Boolean), spouse: d.spouse || null, separated: !!d.separated, createdAt: d.createdAt || 0, masked: true };
    if (nivel === 'nome') { r.name = d.name; r.apelido = d.apelido || ''; r.deceased = !!d.deceased; r.masked = false; r.limited = true; r.links = (d.links || []).filter(l => l.confirmado).map(l => ({ type: l.type, to: l.to, confirmado: true })); }
    return { id: p.id, data: r }; };
  const meu = byId[eu]; let cads = null;
  ((meu && meu.data.links) || []).filter(l => l.confirmado && byId[l.to]).forEach(l => {
    const T = byId[l.to], perto = new Set([T.id, T.data.spouse, T.data.father, T.data.mother].filter(Boolean));
    all.forEach(x => { if (x.data.father === T.id || x.data.mother === T.id) { cads = cads || comCadastro_(); if (filhoLiberado_(byId, T, x, l, cads)) perto.add(x.id); } });
    visiveis_(all, T.id).forEach(id => put(byId[id], perto.has(id) ? 'nome' : 'vazio'));
  });
  all.forEach(p => (p.data.links || []).forEach(l => { if (l.confirmado && vis.has(l.to)) put(p, 'nome'); }));
  return Object.values(out).map(o => limpa(o.p, o.nivel));
}
